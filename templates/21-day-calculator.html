// Shared daily estimate: collection timeline and team totals. function dailyBaseline(ids=team,box=state.box,catalog=window.PS_CATALOG,collectionHours=4,initialEnergy=100,goodCamp=false,threeMeals=true,favoriteBerries=[],areaBonus=0,berryZones={}){ const interval=[1,2,4,8].includes(Number(collectionHours))?Number(collectionHours):4; const mealCount=threeMeals===true?3:threeMeals===false?0:Number.isInteger(threeMeals)&&threeMeals>=0&&threeMeals<=3?threeMeals:3; const mealTicks=[[],[0],[0,72],[0,36,72]][mealCount]; const startingEnergy=Math.max(0,Math.min(150,Math.round(Number(initialEnergy)||0))); const favorites=new Set(favoriteBerries.filter(name=>Object.hasOwn(catalog?.berries||{},name))); const areaFactor=1+Math.max(0,Math.min(200,Number(areaBonus)||0))/100; const energyIntervalFactor=e=>e>=81?.45:e>=61?.52:e>=41?.58:e>=1?.66:1; const speedContext=teamSpeedContext(ids,box,catalog); const periods=[];for(let h=0;h<16;h+=interval)periods.push(Math.min(interval,16-h)); periods.push(8); // No collection during sleep; collect at wake-up. const items=ids.map(id=>box.find(item=>item.id===id)).filter(Boolean); const members=items.map(item=>dailyMemberContext(item,speedContext,items,catalog,startingEnergy)); const valid=members.filter(m=>!m.missing); let elapsedTicks=0; periods.forEach((hours,periodIndex)=>{ let teamRecovery=0,allyRecovery=0,bonusHelps=0; for(const member of valid){ let helps=0; for(let tick=0;tick0||allyRecovery>0)){ // Random ally target is approximated as an equal share; actual targeting favors low energy. for(const member of valid)member.energy=Math.min(150,member.energy+(teamRecovery+allyRecovery/valid.length)*member.stats.energyFactor); } elapsedTicks+=hours*6; }); for(const member of valid){ member.berryEnergyBase=member.berries*berryEnergyAtLevel(member.berryBase,member.level); member.zoneFactor=1+Math.max(0,Math.min(24,Number(berryZones?.[member.berryName])||0))/100; member.berryEnergy=member.berryEnergyBase*(favorites.has(member.berryName)?2:1)*areaFactor*member.zoneFactor; } const special=dailySpecialSkills(valid,items,catalog,areaFactor,favorites,berryZones); const foods=new Map(); valid.forEach(member=>member.foods.forEach((qty,name)=>foods.set(name,(foods.get(name)||0)+qty))); const berryEnergy=valid.reduce((sum,member)=>sum+member.berryEnergy,0); const ordinarySkillEnergyBase=valid.reduce((sum,member)=>sum+(['fixed_energy','variable_energy','fixed_dream_shards_energy'].includes(member.skillEffectType)&&Number.isFinite(member.skillEnergy)?member.skillTriggers*member.skillEnergy:0),0); const skillEnergyBase=ordinarySkillEnergyBase+special.energy/areaFactor; const skillEnergy=ordinarySkillEnergyBase*areaFactor+special.energy; const dreamShards=valid.reduce((sum,member)=>sum+(Number.isFinite(member.skillShards)?member.skillTriggers*member.skillShards:0)+(member.specialShards||0),0); const randomIngredients=valid.reduce((sum,member)=>sum+member.randomIngredients,0); return {members,foods,specialCandy:special.candy,randomIngredients,dreamShards,berryEnergy,skillEnergy,skillEnergyBase,periods,interval,startingEnergy,goodCamp,threeMeals,mealCount,favorites,areaFactor,berryZones}; }
