// Exact additive optimization for each attainable team helping-bonus total. // Individual yields remain unrounded in whistleEstimate; no solo top-five heuristic. function whistleRecommend(box,catalog,options={},target='berry',maxSize=5){ const unique=[...new Map(box.map(item=>[item.id,item])).values()],excluded=[],candidates=[]; if(options.fieldMode==='ex')return {ids:[],excluded,reason:'EXフィールドの補正は未対応です。'}; if((target==='berry'||Array.isArray(target)&&target.some(t=>t.name==='berry'))&&options.favoritesKnown===false)return {ids:[],excluded,reason:'好物きのみを3種類設定してください。'}; for(const item of unique){ const result=whistleEstimate([item.id],unique,catalog,options); if(result.pendingReasons.length||!result.members.length){excluded.push(...result.pendingReasons);continue} const species=speciesFor(item,catalog); const contribution=SUBSKILL_LEVELS.reduce((sum,unlock,i)=>sum+(unlock<=item.level&&SUBSKILLS[item.subskills?.[i]]?.effect==='teamSpeedReduction'?SUBSKILLS[item.subskills[i]].value:0),0); candidates.push({item,species,bonus:Math.round(contribution*1000000)}); } const size=Math.min(maxSize,candidates.length); if(!size)return {ids:[],excluded,reason:'計算できるボックス個体がありません。'}; const sums=Array.from({length:size+1},()=>new Set());sums[0].add(0); for(const c of candidates)for(let n=size;n>0;n--)for(const sum of sums[n-1])sums[n].add(sum+c.bonus); let best=null; for(const total of sums[size]){ const dp=Array.from({length:size+1},()=>new Map());dp[0].set('0|',{score:0,bonus:0,ids:[],special:[]}); for(const c of candidates){ const result=whistleEstimate([c.item.id],unique,catalog,{...options,teamSpeedReduction:total/1000000}); if(result.pendingReasons.length)continue; const score=Array.isArray(target)?target.reduce((sum,t)=>sum+(t.name==='berry'?result.berryEnergy:result.foods.get(t.name)||0)/t.maximum*t.weight,0):target==='berry'?result.berryEnergy:(result.foods.get(target)||0); if(!Number.isFinite(score))continue; for(let n=size;n>0;n--)for(const prev of dp[n-1].values()){ const bonus=prev.bonus+c.bonus;if(bonus>total)continue; const special=c.species.specialTeamLimited===true; if(special&&!specialTeamCompatible(c.species,prev.special))continue; const nextSpecial=special?[...prev.special,c.species]:prev.special; const key=bonus+'|'+nextSpecial.map(p=>p.speciesId||p.no).sort().join(','); const next={score:prev.score+score,bonus,ids:[...prev.ids,c.item.id],special:nextSpecial}; if(!dp[n].has(key)||next.score>dp[n].get(key).score)dp[n].set(key,next); } } for(const result of dp[size].values())if(result.bonus===total&&(!best||result.score>best.score))best=result; } // Fewer than five compatible members can be legitimate in a small special-only box. if(!best&&size>1){ return whistleRecommend(unique,catalog,options,target,size-1); } return best?{ids:best.ids,result:whistleEstimate(best.ids,unique,catalog,options),excluded}:{ids:[],excluded,reason:'編成可能な個体がありません。'}; } function whistleRecommendWeighted(box,catalog,options,targets){ const selected=targets.filter(t=>t.name&&Number.isFinite(t.weight)&&t.weight>0); if(!selected.length)return {ids:[],reason:'項目を1つ以上選んでください。',excluded:[]}; if(new Set(selected.map(t=>t.name)).size!==selected.length)return {ids:[],reason:'同じ項目は重複して選べません。',excluded:[]}; const normalized=[]; for(const t of selected){const best=whistleRecommend(box,catalog,options,t.name);if(!best.ids.length)return best;const maximum=t.name==='berry'?best.result.berryEnergy:best.result.foods.get(t.name)||0;if(maximum<=0)return {ids:[],reason:`${t.name==='berry'?'きのみ':t.name}の獲得見込みがある個体がありません。`,excluded:best.excluded};normalized.push({...t,maximum})} const recommendation=whistleRecommend(box,catalog,options,normalized); recommendation.targets=normalized;return recommendation; }
