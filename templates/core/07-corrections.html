const MAX_POKEMON_LEVEL=70; // Confirmed in official Ver.3.6.0 notice, 2026-06-25. const SUBSKILL_LEVELS=[10,25,50,70,80]; // Individual corrections are deliberately separate from daily yield (energy, collection and camp are not fixed here). const NATURE_ROWS=window.PS_CATALOG.natures.rows; const NATURE_AXES=window.PS_CATALOG.natures.axes; const NATURE_EFFECTS=Object.fromEntries(NATURE_ROWS.flatMap((row,up)=>row.map((name,down)=>[name,up===down?{}:{up:NATURE_AXES[up],down:NATURE_AXES[down]}]))); const SUBSKILLS=Object.fromEntries(window.PS_CATALOG.subskills.map(skill=>[skill.name,skill])); const SUBSKILL_NAMES=window.PS_CATALOG.subskills.map(skill=>skill.name); function remainingEvolutionCount(value,seen=new Set(),catalog=window.PS_CATALOG){ const p=typeof value==='object'?value:catalog?.pokemon?.[value];if(!p)throw Error('進化先のデータが未登録です。'); const no=p.speciesId||p.no; if(seen.has(no))throw Error('進化データを確認してください。'); if(p.detailPreviewOnly){ const edges=p.evolutionTransitions||[]; if(edges.some(e=>!e.transitionEnabled))throw Error('この姿の進化回数は未確認です。'); return edges.length?Math.max(...edges.map(e=>1+remainingEvolutionCount(window.PS_FORMS.resolve(e.targetSpeciesId),new Set([...seen,no]),catalog))):0; } const targets=(Array.isArray(p.evolution)?p.evolution:[p.evolution]).filter(Boolean).map(e=>e.toNo); return targets.length?Math.max(...targets.map(target=>1+remainingEvolutionCount(target,new Set([...seen,no]),catalog))):0; } function individualRibbon(item,p){ if(!item.ribbonHours)return {capacity:0,time:1}; try{return window.PS_SWAP_ENGINE.ribbon(item.ribbonHours,remainingEvolutionCount(p))} catch{return {capacity:null,time:null}} } // Capacity follows the current species' stage, regardless of capture history. const speciesEvolutionStageCache=new WeakMap(); function speciesEvolutionStage(p,catalog=window.PS_CATALOG){ if(!p)return null; const key=value=>value.speciesId||`${String(value.no).padStart(4,'0')}_default`; let stages=speciesEvolutionStageCache.get(catalog); if(!stages){ const records=new Map([...Object.values(catalog.pokemon||{}),...(catalog.forms?.species||[])].map(value=>[key(value),value])); const parents=new Map(); for(const [id,value] of records){ const edges=[...(Array.isArray(value.evolution)?value.evolution:[value.evolution]).filter(edge=>edge?.toNo).map(edge=>`${String(edge.toNo).padStart(4,'0')}_default`), ...(value.evolutionTransitions||[]).filter(edge=>edge.targetSpeciesId&&edge.transitionEnabled!==false).map(edge=>edge.targetSpeciesId)]; for(const target of edges){if(!parents.has(target))parents.set(target,[]);parents.get(target).push(id)} } stages=new Map(); const visit=(id,seen=new Set())=>{ if(stages.has(id))return stages.get(id);if(seen.has(id))return null; const incoming=parents.get(id)||[],value=records.get(id),normal=value?`${String(value.no).padStart(4,'0')}_default`:id; const previous=incoming.map(parent=>visit(parent,new Set([...seen,id]))); const stage=previous.some(n=>n==null)?null:previous.length?1+Math.max(...previous):normal!==id&&records.has(normal)?visit(normal,new Set([...seen,id])):0; stages.set(id,stage);return stage; }; for(const id of records.keys())visit(id); speciesEvolutionStageCache.set(catalog,stages); } const id=key(p),normal=`${String(p.no).padStart(4,'0')}_default`; return stages.has(id)?stages.get(id):stages.has(normal)?stages.get(normal):0; } function individualCorrections(item,p,teamSpeedReduction=0){ const effect=item.mint?{}:NATURE_EFFECTS[item.nature]||{}; const active=SUBSKILL_LEVELS.map((n,i)=>n<=item.level&&(!p?.mythicalSettings||item.mythicalState?.subskillUnlocked?.[i]===true)?item.subskills?.[i]:null).filter(Boolean); const sum=kind=>active.reduce((total,name)=>total+(SUBSKILLS[name]?.effect===kind?SUBSKILLS[name].value||0:0),0); const natureFactor=axis=>effect.up===axis?window.PS_CATALOG.natures.factors[axis].up:effect.down===axis?window.PS_CATALOG.natures.factors[axis].down:1; const soloSpeedReduction=Math.min(.35,sum('speedReduction')); const speedReduction=Math.min(.35,sum('speedReduction')+teamSpeedReduction); const ribbon=individualRibbon(item,p); const speedBase=Number.isFinite(p?.help)&&ribbon.time!==null?p.help*(1-(item.level-1)*.002)*natureFactor('speed')*ribbon.time:null; const soloSpeed=speedBase==null?null:speedBase*(1-soloSpeedReduction); const speed=speedBase==null?null:speedBase*(1-speedReduction); const foodFactor=natureFactor('food')*(1+sum('foodRateBonus')); const skillFactor=natureFactor('skill')*(1+sum('skillRateBonus')); const evolutionStage=speciesEvolutionStage(p),evolutionCarry=evolutionStage==null?null:5*evolutionStage; const carry=Number.isFinite(p?.carry)&&ribbon.capacity!==null&&evolutionCarry!==null?p.carry+evolutionCarry+sum('carryBonus')+ribbon.capacity:null; const food=Number.isFinite(p?.foodRate)?Math.min(100,p.foodRate*foodFactor):null; const skill=Number.isFinite(p?.skillRate)?Math.min(100,p.skillRate*skillFactor):null; const unknown=[...active.filter(name=>!SUBSKILL_NAMES.includes(name)),...(!item.mint&&item.nature&&!NATURE_EFFECTS[item.nature]?[item.nature]:[])]; return {speed,soloSpeed,carry,food,skill,foodFactor,skillFactor,active,unknown,ribbon, nature:effect,berryQty:Number.isFinite(p?.berryQty)?p.berryQty+sum('berryQtyBonus'):null, teamBonus:sum('teamSpeedReduction')>0,energyFactor:natureFactor('energy'),expFactor:natureFactor('exp'), breakdown:{levelFactor:1-(item.level-1)*.002,natureSpeed:natureFactor('speed'),natureFood:natureFactor('food'),natureSkill:natureFactor('skill'), soloSpeedReduction,speedReduction,teamSpeedReduction,foodBonus:sum('foodRateBonus'),skillBonus:sum('skillRateBonus'), evolutionStage,evolutionCarry,subskillCarry:sum('carryBonus')}}; } function teamSpeedContext(ids=team,box=state.box,catalog=window.PS_CATALOG){ const selected=ids.map(id=>box.find(item=>item.id===id)).filter(Boolean); const contribution=item=>SUBSKILL_LEVELS.reduce((sum,unlock,i)=>sum+(unlock<=item.level&&(!speciesFor(item,catalog)?.mythicalSettings||item.mythicalState?.subskillUnlocked?.[i]===true)&&SUBSKILLS[item.subskills?.[i]]?.effect==='teamSpeedReduction'?SUBSKILLS[item.subskills[i]].value:0),0); const bonus=selected.reduce((sum,item)=>sum+contribution(item),0); const members=new Map(selected.map(item=>[item.id,individualCorrections(item,speciesFor(item,catalog),bonus)])); return {bonus,bonusCount:selected.filter(item=>contribution(item)>0).length,members};
