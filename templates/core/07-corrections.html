const MAX_POKEMON_LEVEL=70; // Confirmed in official Ver.3.6.0 notice, 2026-06-25. const SUBSKILL_LEVELS=[10,25,50,70,80]; // Individual corrections are deliberately separate from daily yield (energy, collection and camp are not fixed here). const NATURE_ROWS=window.PS_CATALOG.natures.rows; const NATURE_AXES=window.PS_CATALOG.natures.axes; const NATURE_EFFECTS=Object.fromEntries(NATURE_ROWS.flatMap((row,up)=>row.map((name,down)=>[name,up===down?{}:{up:NATURE_AXES[up],down:NATURE_AXES[down]}]))); const SUBSKILLS=Object.fromEntries(window.PS_CATALOG.subskills.map(skill=>[skill.name,skill])); const SUBSKILL_NAMES=window.PS_CATALOG.subskills.map(skill=>skill.name); function remainingEvolutionCount(value,seen=new Set(),catalog=window.PS_CATALOG){ const p=typeof value==='object'?value:catalog?.pokemon?.[value];if(!p)throw Error('進化先のデータが未登録です。'); const no=p.speciesId||p.no; if(seen.has(no))throw Error('進化データを確認してください。'); if(p.detailPreviewOnly){ const edges=p.evolutionTransitions||[]; if(edges.some(e=>!e.transitionEnabled))throw Error('この姿の進化回数は未確認です。'); return edges.length?Math.max(...edges.map(e=>1+remainingEvolutionCount(window.PS_FORMS.resolve(e.targetSpeciesId),new Set([...seen,no]),catalog))):0; } const targets=(Array.isArray(p.evolution)?p.evolution:[p.evolution]).filter(Boolean).map(e=>e.toNo); return targets.length?Math.max(...targets.map(target=>1+remainingEvolutionCount(target,new Set([...seen,no]),catalog))):0; } function individualRibbon(item,p){ if(!item.ribbonHours)return {capacity:0,time:1}; try{return window.PS_SWAP_ENGINE.ribbon(item.ribbonHours,remainingEvolutionCount(p))} catch{return {capacity:null,time:null}} } function individualCorrections(item,p,teamSpeedReduction=0){ const effect=item.mint?{}:NATURE_EFFECTS[item.nature]||{}; const active=SUBSKILL_LEVELS.map((n,i)=>n<=item.level&&(!p?.mythicalSettings||item.mythicalState?.subskillUnlocked?.[i]===true)?item.subskills?.[i]:null).filter(Boolean); const sum=kind=>active.reduce((total,name)=>total+(SUBSKILLS[name]?.effect===kind?SUBSKILLS[name].value||0:0),0); const natureFactor=axis=>effect.up===axis?window.PS_CATALOG.natures.factors[axis].up:effect.down===axis?window.PS_CATALOG.natures.factors[axis].down:1; const soloSpeedReduction=Math.min(.35,sum('speedReduction')); const speedReduction=Math.min(.35,sum('speedReduction')+teamSpeedReduction); const ribbon=individualRibbon(item,p); const speedBase=Number.isFinite(p?.help)&&ribbon.time!==null?p.help*(1-(item.level-1)*.002)*natureFactor('speed')*ribbon.time:null; const soloSpeed=speedBase==null?null:speedBase*(1-soloSpeedReduction); const speed=speedBase==null?null:speedBase*(1-speedReduction); const foodFactor=natureFactor('food')*(1+sum('foodRateBonus')); const skillFactor=natureFactor('skill')*(1+sum('skillRateBonus')); const carry=Number.isFinite(p?.carry)&&ribbon.capacity!==null?p.carry+5*([0,1,2].includes(item.evolutionCount)?item.evolutionCount:0)+sum('carryBonus')+ribbon.capacity:null; const food=Number.isFinite(p?.foodRate)?Math.min(100,p.foodRate*foodFactor):null; const skill=Number.isFinite(p?.skillRate)?Math.min(100,p.skillRate*skillFactor):null; const unknown=[...active.filter(name=>!SUBSKILL_NAMES.includes(name)),...(!item.mint&&item.nature&&!NATURE_EFFECTS[item.nature]?[item.nature]:[])]; return {speed,soloSpeed,carry,food,skill,foodFactor,skillFactor,active,unknown,ribbon, nature:effect,berryQty:Number.isFinite(p?.berryQty)?p.berryQty+sum('berryQtyBonus'):null, teamBonus:sum('teamSpeedReduction')>0,energyFactor:natureFactor('energy'),expFactor:natureFactor('exp')}; } function teamSpeedContext(ids=team,box=state.box,catalog=window.PS_CATALOG){ const selected=ids.map(id=>box.find(item=>item.id===id)).filter(Boolean); const contribution=item=>SUBSKILL_LEVELS.reduce((sum,unlock,i)=>sum+(unlock<=item.level&&(!speciesFor(item,catalog)?.mythicalSettings||item.mythicalState?.subskillUnlocked?.[i]===true)&&SUBSKILLS[item.subskills?.[i]]?.effect==='teamSpeedReduction'?SUBSKILLS[item.subskills[i]].value:0),0); const bonus=selected.reduce((sum,item)=>sum+contribution(item),0); const members=new Map(selected.map(item=>[item.id,individualCorrections(item,speciesFor(item,catalog),bonus)])); return {bonus,bonusCount:selected.filter(item=>contribution(item)>0).length,members};
