function recipeCollectionPanel(recipe){ const section=document.createElement('section');section.className='psg-recipe-team-time'; const add=(tag,copy)=>{const el=document.createElement(tag);if(copy!==undefined)el.textContent=copy;section.append(el);return el}; add('h3','現在のチームで集める時間'); const label=add('label','作る食数 '),meals=document.createElement('select');meals.setAttribute('aria-label','収集目安の食数'); for(const n of [1,3]){const option=new Option(`${n}食分`,String(n));meals.add(option)}label.append(meals); const stock=readRecipeStock(),stockInputs=new Map(),inputs=add('div');inputs.className='psg-recipe-stock-inputs'; const output=add('div');output.setAttribute('aria-live','polite'); const render=()=>{ output.replaceChildren(); if([...inputs.querySelectorAll('input')].some(input=>!input.checkValidity())){output.textContent='在庫は0以上の整数で入力してください';return} const line=copy=>{const p=document.createElement('p');p.className='note';p.textContent=copy;output.append(p)}; try{ const forecast=dailyCurrentForecast(),result=recipeCollectionEstimate(recipe.ingredients,forecast,stock,Number(meals.value)); line(`回収${forecast.interval||daySettings.collectionHours}時間・起床げんき${teamInitialEnergy}・食事${teamMeals}回。チーム画面の現在条件を使用。`); const title=document.createElement('strong');title.textContent=result.status==='available'?'在庫で食材が揃います':result.status==='unconfirmed'?'収集時間は未確認':result.status==='unavailable'?'この編成の種類別予想では集まりません':`約${window.PS_NUMBERS.display(result.hours)}時間`;output.append(title); for(const row of result.rows)line(`${row.name}：あと${window.PS_NUMBERS.display(row.deficit)}個${row.status==='unconfirmed'?'（日産未確認）':row.status==='not_produced'?'（種類別日産0）':row.deficit===0?'（在庫あり）':`／約${window.PS_NUMBERS.display(row.hours)}時間`}`); if(result.bottlenecks.length)line('最後に揃う食材：'+result.bottlenecks.join('・')); const capacity=cookingContext().capacity,required=recipe.ingredients.reduce((sum,r)=>sum+r.qty,0); line(capacity===null?'鍋容量は未設定':required>capacity?`1食の鍋容量不足：あと${window.PS_NUMBERS.display(required-capacity)}個`:`1食の必要数${required}個は現在の鍋容量内`); line('平均収集量からの目安（推定）。実際の回収時刻・成功確率・朝昼晩に調理できるかは判定しません。種類未指定のランダム食材は加算しません。在庫は消費・獲得後に手動更新してください。'); }catch(e){line(e.message)} }; for(const item of recipe.ingredients){ const label=document.createElement('label');label.textContent=item.name+' 在庫 ';const input=document.createElement('input');input.type='number';input.min='0';input.step='1';input.inputMode='numeric';input.value=String(stock[item.name]||0);input.setAttribute('aria-label',item.name+'の在庫数');label.append(input);inputs.append(label);stockInputs.set(item.name,input); input.oninput=()=>{const value=Number(input.value);input.setCustomValidity(input.value.trim()===''||!Number.isInteger(value)||value<0?'0以上の整数を入力してください':'');if(!input.checkValidity()){output.textContent='在庫は0以上の整数で入力してください';return}stock[item.name]=value;try{const current=readRecipeStock();localStorage.setItem(RECIPE_STOCK_KEY,JSON.stringify({...current,[item.name]:value}))}catch{output.textContent='在庫を保存できませんでした';return}render()}; } meals.onchange=render;render(); section.refreshRecipeTime=()=>{const current=readRecipeStock();for(const [name,input] of stockInputs){stock[name]=current[name]??0;input.value=String(stock[name]);input.setCustomValidity('')}render()}; return section; }
