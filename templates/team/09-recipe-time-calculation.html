// Convert the unrounded whole-team daily forecast into a constant-flow collection estimate. function recipeCollectionEstimate(requirements,forecast,stock={},meals=1){ if(![1,3].includes(meals))throw Error('食数を確認してください'); const ids=forecast.members.map(m=>m.id); if(ids.length>5||new Set(ids).size!==ids.length)throw Error('同じ個体を重複して編成できません'); const incomplete=!ids.length||forecast.pendingReason||forecast.members.some(m=>m.missing); const rows=requirements.map(item=>{ const available=stock[item.name]??0; if(!Number.isFinite(available)||available<0||!Number.isFinite(item.qty)||item.qty<0)throw Error('在庫・必要数を確認してください'); const deficit=Math.max(0,item.qty*meals-available); // An absent food is zero only after every team member has a complete forecast. const daily=incomplete?null:(forecast.foods.has(item.name)?forecast.foods.get(item.name):0); if(daily!==null&&(!Number.isFinite(daily)||daily<0))throw Error('日産量を確認してください'); const status=deficit===0?'available':daily===null?'unconfirmed':daily===0?'not_produced':'estimate'; return {name:item.name,deficit,daily,status,hours:deficit===0?0:status==='estimate'?24*deficit/daily:null}; }); const unknown=rows.some(r=>r.status==='unconfirmed'),unavailable=rows.some(r=>r.status==='not_produced'); const hours=unknown||unavailable?null:Math.max(0,...rows.map(r=>r.hours)); return {rows,hours,status:unknown?'unconfirmed':unavailable?'unavailable':hours===0?'available':'estimate',bottlenecks:hours>0?rows.filter(r=>Math.abs(r.hours-hours)<=1e-12*Math.max(1,hours)).map(r=>r.name):[]}; }
